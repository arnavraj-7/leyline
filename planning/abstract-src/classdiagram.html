classDiagram
    direction LR
    class NodeId {
        +Int value
        +BROADCAST$
    }
    class MessageId {
        +Long value
    }
    class Packet {
        +MessageId id
        +PacketType type
        +Priority priority
        +NodeId origin
        +NodeId destination
        +Int ttl
        +Long createdAt
        +List~NodeId~ path
        +ByteArray payload
        +hopCount() Int
    }
    class PacketCodec {
        <<object>>
        +encode(Packet) ByteArray
        +decode(ByteArray) Packet
    }
    class SeenCache {
        -capacity Int
        -expiryMillis Long
        +markIfNew(MessageId) Boolean
    }
    class Router {
        -self NodeId
        +originate(payload, destination, priority, ttl) Packet
        +onReceive(Packet) RouteDecision
    }
    class RouteDecision {
        <<sealed>>
        Drop(reason)
        Accept(deliver, forward)
    }
    class ForwardQueue {
        +offer(Packet)
        +poll() Packet
    }
    class Chunker {
        +split(ByteArray, mtu) List~Chunk~
    }
    class Reassembler {
        +accept(Chunk) ByteArray?
    }
    class Outbox {
        <<interface>>
        +save(Packet)
        +pending() List~Packet~
        +markSent(MessageId, NodeId)
    }
    class Transport {
        <<interface>>
        +neighbours() Flow
        +send(NodeId, ByteArray)
    }
    class MeshNode {
        +send(text, priority)
        +onBytes(ByteArray, NodeId)
    }
    class BleTransport
    class ConnectionPool
    class GattServerManager
    class GattClientManager
    class RoomOutbox
    class MeshService
    class ChatViewModel
